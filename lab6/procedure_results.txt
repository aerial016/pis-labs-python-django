CREATE PROCEDURE create_article(

    IN p_title VARCHAR(200),

    IN p_text TEXT,

    IN p_author_id INTEGER,

    OUT article_id INTEGER

)

LANGUAGE plpgsql

AS $$

BEGIN

    INSERT INTO articles_article (title, text, created_date, author_id)

    VALUES (p_title, p_text, CURRENT_DATE, p_author_id)

    RETURNING id INTO article_id;

END;

$$;

CREATE PROCEDURE rename_article(

    IN p_article_id INTEGER,

    IN p_title VARCHAR(200)

)

LANGUAGE plpgsql

AS $$

BEGIN

    UPDATE articles_article SET title = p_title WHERE id = p_article_id;

END;

$$;
